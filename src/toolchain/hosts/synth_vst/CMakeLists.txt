cmake_minimum_required(VERSION 3.25)
project(osprey_synth_vst LANGUAGES C CXX)
if(APPLE)
    enable_language(OBJCXX)
endif()
if(NOT IS_DIRECTORY "${VST3_SDK}")
    message(FATAL_ERROR "VST3_SDK must name the pinned SDK checkout")
endif()
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK}" sdk EXCLUDE_FROM_ALL)
add_executable(osprey-vst-worker main.cpp processor.cpp stream.cpp descriptor.cpp stream_text.cpp audio_device.cpp
    "${VST3_SDK}/public.sdk/source/common/memorystream.cpp"
    "${VST3_SDK}/public.sdk/source/vst/hosting/plugprovider.cpp")
target_compile_features(osprey-vst-worker PRIVATE cxx_std_20)
# Keep our warnings fatal while treating unmodified upstream headers as external.
target_include_directories(osprey-vst-worker SYSTEM PRIVATE "${VST3_SDK}")
target_link_libraries(osprey-vst-worker PRIVATE sdk_hosting)
if(APPLE)
    target_sources(osprey-vst-worker PRIVATE editor_mac.mm "${VST3_SDK}/public.sdk/source/vst/hosting/module_mac.mm")
    set_source_files_properties(editor_mac.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    target_link_libraries(osprey-vst-worker PRIVATE "-framework Cocoa")
    set_source_files_properties("${VST3_SDK}/public.sdk/source/vst/hosting/module_mac.mm" PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    target_link_libraries(osprey-vst-worker PRIVATE "-framework CoreFoundation" "-framework Foundation" "-framework AudioToolbox")
elseif(WIN32)
    target_sources(osprey-vst-worker PRIVATE "${VST3_SDK}/public.sdk/source/vst/hosting/module_win32.cpp")
else()
    target_sources(osprey-vst-worker PRIVATE "${VST3_SDK}/public.sdk/source/vst/hosting/module_linux.cpp")
    target_link_libraries(osprey-vst-worker PRIVATE dl)
endif()
if(MSVC)
    target_compile_options(osprey-vst-worker PRIVATE /W4 /WX)
else()
    target_compile_options(osprey-vst-worker PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME rejects_missing_plugin COMMAND osprey-vst-worker scan /definitely/missing/plugin.vst3)
set_tests_properties(rejects_missing_plugin PROPERTIES WILL_FAIL TRUE TIMEOUT 10)
# Real SDK fixtures run through the same disposable process used for third-party modules.
get_target_property(delay_bundle adelay SMTG_PLUGIN_PACKAGE_PATH)
get_target_property(mda_bundle mda-vst3 SMTG_PLUGIN_PACKAGE_PATH)
add_test(NAME scan_effect COMMAND osprey-vst-worker scan "${delay_bundle}")
set_tests_properties(scan_effect PROPERTIES PASS_REGULAR_EXPRESSION "ADelay" TIMEOUT 10)
add_test(NAME scan_instruments COMMAND osprey-vst-worker scan "${mda_bundle}")
set_tests_properties(scan_instruments PROPERTIES PASS_REGULAR_EXPRESSION "mda JX10" TIMEOUT 10)
add_test(NAME process_effect COMMAND osprey-vst-worker probe "${delay_bundle}" 0CDBB66985D548A9BFD8371909D24BB3)
add_test(NAME process_instrument COMMAND osprey-vst-worker probe "${mda_bundle}" 5653544D44416A6D6461206A78313000)
set_tests_properties(process_effect process_instrument PROPERTIES TIMEOUT 20)
add_test(NAME rejects_missing_class COMMAND osprey-vst-worker probe "${mda_bundle}" 00000000000000000000000000000000)
set_tests_properties(rejects_missing_class PROPERTIES WILL_FAIL TRUE TIMEOUT 10)
add_test(NAME state_automation_effect COMMAND osprey-vst-worker verify-state "${delay_bundle}" 0CDBB66985D548A9BFD8371909D24BB3)
add_test(NAME state_automation_instrument COMMAND osprey-vst-worker verify-state "${mda_bundle}" 5653544D44416A6D6461206A78313000)
set_tests_properties(state_automation_effect state_automation_instrument PROPERTIES TIMEOUT 20)

add_test(NAME controller_program_restart COMMAND osprey-vst-worker verify-restart "${mda_bundle}" 5653544D44416A6D6461206A78313000)
set_tests_properties(controller_program_restart PROPERTIES PASS_REGULAR_EXPRESSION "Received program parameter restart" TIMEOUT 10)

add_test(NAME effect_state_session COMMAND node "${CMAKE_CURRENT_SOURCE_DIR}/state-session.mjs" $<TARGET_FILE:osprey-vst-worker> "${delay_bundle}" 0CDBB66985D548A9BFD8371909D24BB3 2 100)
add_test(NAME instrument_state_session COMMAND node "${CMAKE_CURRENT_SOURCE_DIR}/state-session.mjs" $<TARGET_FILE:osprey-vst-worker> "${mda_bundle}" 5653544D44416A6D6461206A78313000 0 0)
set_tests_properties(effect_state_session instrument_state_session PROPERTIES TIMEOUT 20)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME stream_journeys COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/build.py" --stream-test)
set_tests_properties(stream_journeys PROPERTIES TIMEOUT 60)

if(APPLE)
    add_test(NAME audio_device_journeys COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/build.py" --device-test)
    add_test(NAME audio_device_cadence COMMAND node "${CMAKE_CURRENT_SOURCE_DIR}/device-cadence.mjs" $<TARGET_FILE:osprey-vst-worker>)
    set_tests_properties(audio_device_journeys audio_device_cadence PROPERTIES TIMEOUT 20 RESOURCE_LOCK audio_device)
endif()
